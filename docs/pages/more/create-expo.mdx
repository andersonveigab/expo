// Estrutura inicial do app Move Cidade (React Native + Expo + OpenRouteService)

import React, { useState } from "react";
import { View, Text, TextInput, TouchableOpacity, StyleSheet, Alert } from "react-native";
import MapView, { Marker } from "react-native-maps";

const ORS_API_KEY = "5b3ce3597851110001cf62480bbfd6d799c74196a783963e56fb4ced"; // Substitua pela sua chave da OpenRouteService

export default function App() {
  const [pickup, setPickup] = useState("");
  const [destination, setDestination] = useState("");
  const [distanceKm, setDistanceKm] = useState(0);

  const calcularPrecoCorrida = (distancia) => {
    if (distancia <= 4) return 8;
    return 8 + (distancia - 4) * 2;
  };

  const pedirViagem = async () => {
    if (!pickup || !destination) {
      Alert.alert("Erro", "Informe partida e destino");
      return;
    }

    try {
      const coordsOrigem = await buscarCoordenadas(pickup);
      const coordsDestino = await buscarCoordenadas(destination);

      const distanciaKm = await calcularDistancia(coordsOrigem, coordsDestino);
      setDistanceKm(distanciaKm);

      const preco = calcularPrecoCorrida(distanciaKm);
      const taxaApp = preco * 0.10;
      const motoristaRecebe = preco - taxaApp;

      Alert.alert(
        "Resumo da Corrida",
        `Distância estimada: ${distanciaKm.toFixed(2)} km\nPreço da corrida: R$ ${preco.toFixed(2)}\nTaxa do app: R$ ${taxaApp.toFixed(2)}\nMotorista recebe: R$ ${motoristaRecebe.toFixed(2)}`
      );
    } catch (error) {
      console.error(error);
      Alert.alert("Erro", "Não foi possível calcular a rota");
    }
  };

  const buscarCoordenadas = async (endereco) => {
    const response = await fetch(
      `https://api.openrouteservice.org/geocode/search?api_key=${ORS_API_KEY}&text=${encodeURIComponent(endereco)}&boundary.country=BR`
    );
    const data = await response.json();
    if (data.features.length === 0) throw new Error("Endereço não encontrado");
    return data.features[0].geometry.coordinates;
  };

  const calcularDistancia = async (coordOrigem, coordDestino) => {
    const response = await fetch("https://api.openrouteservice.org/v2/directions/driving-car/geojson", {
      method: "POST",
      headers: {
        "Content-Type": "application/json",
        Authorization: ORS_API_KEY,
      },
      body: JSON.stringify({
        coordinates: [coordOrigem, coordDestino],
      }),
    });
    const data = await response.json();
    const distanciaMetros = data.features[0].properties.summary.distance;
    return distanciaMetros / 1000; // converter para km
  };

  return (
    <View style={styles.container}>
      <Text style={styles.logo}>Move Cidade</Text>
      <Text style={styles.welcome}>Bem-vindo!</Text>
      <MapView
        style={styles.map}
        initialRegion={{
          latitude: -22.1256,
          longitude: -51.3889,
          latitudeDelta: 0.01,
          longitudeDelta: 0.01,
        }}
      >
        <Marker
          coordinate={{ latitude: -22.1256, longitude: -51.3889 }}
          title="Você está aqui"
        />
      </MapView>
      <TextInput
        style={styles.input}
        placeholder="Informe o local de partida"
        value={pickup}
        onChangeText={setPickup}
      />
      <TextInput
        style={styles.input}
        placeholder="Informe o destino"
        value={destination}
        onChangeText={setDestination}
      />
      <TouchableOpacity style={styles.button} onPress={pedirViagem}>
        <Text style={styles.buttonText}>Pedir viagem</Text>
      </TouchableOpacity>
      <Text style={styles.loginText}>Entrar ou Registrar</Text>
    </View>
  );
}

  },

  },
  button: {
    backgroundColor: '#000',

    borderRadius: 8,
    marginTop: 15,
    alignItems: 'center',
  },
  buttonText: {
    color: '#fff',
    fontWeight: 'bold',
  },
  loginText: {
    textAlign: 'center',
    marginTop: 20,
    color: '#555',
  },
});
